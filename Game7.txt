      if (window.location.protocol === 'file:') {
        hostUrl = 'http://192.168.1.100:5500/index.html'; // 占位局域网示范
      }
      document.getElementById('hostOverrideInput').value = hostUrl;
      refreshQRImage(hostUrl);
    }

    function updateQRCodeFromInput() {
      const customUrl = document.getElementById('hostOverrideInput').value.trim();
      if (!customUrl) return;
      refreshQRImage(customUrl);
    }

    function refreshQRImage(baseUrl) {
      const cleanUrl = baseUrl.split('?')[0];
      const targetMobileUrl = `${cleanUrl}?role=mobile&room=${roomId}`;
      document.getElementById('roomCodeDisplay').innerText = roomId;
      document.getElementById('qrCodeImg').src = `https://api.qrserver.com/v1/create-qr-code/?size=220x220&data=${encodeURIComponent(targetMobileUrl)}`;
      document.getElementById('mobileDirectLink').href = targetMobileUrl;
    }

    function openQRModal() {
      SFX.tap();
      document.getElementById('qrModal').classList.remove('hidden');
    }
    function closeQRModal() {
      SFX.tap();
      document.getElementById('qrModal').classList.add('hidden');
    }

    function renderScreen() {
      if (questions.length === 0) return;
      const cur = questions[gameState.qIndex];

      // 规范汉字展示
      document.getElementById('centerHanziText').innerText = cur.hanzi;
      document.getElementById('roundIndicator').innerText = `ROUND ${gameState.qIndex + 1} / ${questions.length}`;
      document.getElementById('revealPinyin').innerText = cur.pinyin;
      document.getElementById('revealMeaning').innerText = `释义: ${cur.meaning}`;
      document.getElementById('revealCollocation').innerText = `常用搭配: ${cur.collocation}`;

      if (gameState.revealed) {
        document.getElementById('scaffoldDrawer').classList.remove('hidden');
      } else {
        document.getElementById('scaffoldDrawer').classList.add('hidden');
      }

      renderTeamState('red', gameState.redBalloons, 'redHealthSlots', 'redBalloonCluster', 'redPilotAvatar', 'redPilotZone');
      renderTeamState('blue', gameState.blueBalloons, 'blueHealthSlots', 'blueBalloonCluster', 'bluePilotAvatar', 'bluePilotZone');
    }

    function renderTeamState(team, count, healthId, clusterId, avatarId, zoneId) {
      const healthEl = document.getElementById(healthId);
      const clusterEl = document.getElementById(clusterId);
      const avatarEl = document.getElementById(avatarId);
      const zoneEl = document.getElementById(zoneId);
      const color = team === 'red' ? '#FF385C' : '#2B7FFF';

      healthEl.innerHTML = '';
      for (let i = 0; i < gameState.maxBalloons; i++) {
        if (i < count) {
          healthEl.innerHTML += getBalloonSVG(color, 20);
        } else {
          healthEl.innerHTML += `<div class="w-5 h-6 rounded-full border-2 border-dashed border-black/40"></div>`;
        }
      }

      clusterEl.innerHTML = '';
      for (let i = 0; i < count; i++) {
        clusterEl.innerHTML += getBalloonSVG(color, 38);
      }

      avatarEl.innerHTML = getDetailedPilotSVG(team);
      if (count === 1) avatarEl.classList.add('animate-panic');
      else avatarEl.classList.remove('animate-panic');

      const dropPx = (gameState.maxBalloons - count) * 48;
      if (count > 0) {
        zoneEl.style.transform = `translateY(${dropPx}px)`;
        zoneEl.style.opacity = '1';
      } else {
        zoneEl.style.transform = `translateY(280px) scale(0.3) rotate(45deg)`;
        zoneEl.style.opacity = '0';
      }
    }

    function startTimer() {
      clearInterval(gameState.timerInterval);
      gameState.timer = gameState.maxTime;
      gameState.timerActive = true;
      gameState.status = 'ACTIVE';
      updateTimerUI();

      broadcastPayload({ type: 'SYNC_STATE', status: 'ACTIVE', buzzedTeam: null });

      gameState.timerInterval = setInterval(() => {
        if (!gameState.timerActive) return;
        gameState.timer--;
        updateTimerUI();

        if (gameState.timer <= 0) {
          clearInterval(gameState.timerInterval);
          gameState.timerActive = false;
          gameState.status = 'WAITING';
          SFX.wrong();
          document.getElementById('timerNumber').innerText = '00s';
          broadcastPayload({ type: 'SYNC_STATE', status: 'WAITING', buzzedTeam: null });
        }
      }, 1000);
    }

    function pauseTimer() {
      gameState.timerActive = false;
    }

    function resetTimer() {
      SFX.tap();
      unlockBuzz();
      startTimer();
    }

    function updateTimerUI() {
      const numEl = document.getElementById('timerNumber');
      const barEl = document.getElementById('timerProgressBar');
      numEl.innerText = `${gameState.timer < 10 ? '0' : ''}${gameState.timer}s`;
      const pct = (gameState.timer / gameState.maxTime) * 100;
      barEl.style.width = `${pct}%`;

      if (gameState.timer <= 3) {
        barEl.className = 'absolute bottom-0 left-0 h-2 bg-red-600 transition-all duration-1000 animate-pulse';
        numEl.classList.add('text-red-500');
      } else {
        barEl.className = 'absolute bottom-0 left-0 h-2 bg-emerald-500 transition-all duration-1000';
        numEl.classList.remove('text-red-500');
      }
    }

    function triggerBuzz(team) {
      if (gameState.buzzedTeam !== null || gameState.status === 'WAITING' || gameState.status === 'GAMEOVER') return;
      gameState.buzzedTeam = team;
      gameState.status = 'BUZZED';
      pauseTimer();
      SFX.buzz();

      const banner = document.getElementById('buzzOverlayBanner');
      const tag = document.getElementById('buzzSquadTag');
      banner.classList.remove('hidden');

      if (team === 'red') {
        tag.innerText = '🔴 RED SQUAD BUZZED!';
        tag.style.backgroundColor = '#FF385C';
        document.getElementById('redBuzzerBtn').classList.add('sc-btn-active');
      } else {
        tag.innerText = '🔵 BLUE SQUAD BUZZED!';
        tag.style.backgroundColor = '#2B7FFF';
        document.getElementById('blueBuzzerBtn').classList.add('sc-btn-active');
      }

      broadcastPayload({ type: 'SYNC_STATE', status: 'BUZZED', buzzedTeam: team });
    }

    function unlockBuzz() {
      gameState.buzzedTeam = null;
      document.getElementById('buzzOverlayBanner').classList.add('hidden');
      document.getElementById('redBuzzerBtn').classList.remove('sc-btn-active');
      document.getElementById('blueBuzzerBtn').classList.remove('sc-btn-active');
    }

    function handleVerdict(isCorrect) {
      if (!gameState.buzzedTeam) return;
      const answeringTeam = gameState.buzzedTeam;
      const opponentTeam = answeringTeam === 'red' ? 'blue' : 'red';

      if (isCorrect) {
        SFX.correct();
        deductBalloon(opponentTeam);
      } else {
        SFX.wrong();
        deductBalloon(answeringTeam);
      }

      unlockBuzz();
      gameState.revealed = true;
      renderScreen();
    }

    function deductBalloon(team) {
      SFX.pop();
      if (team === 'red') {
        gameState.redBalloons = Math.max(0, gameState.redBalloons - 1);
        if (gameState.redBalloons === 0) triggerMonsterExecution('red');
      } else {
        gameState.blueBalloons = Math.max(0, gameState.blueBalloons - 1);
        if (gameState.blueBalloons === 0) triggerMonsterExecution('blue');
      }
      renderScreen();
    }

    function triggerMonsterExecution(defeatedTeam) {
      gameState.status = 'GAMEOVER';
      pauseTimer();
      SFX.slideDrop();

      setTimeout(() => {
        SFX.chomp();
        const fish = document.getElementById('monsterFishContainer');
        fish.style.transform = 'translateY(-40px) scale(1.2)';
        setTimeout(() => {
          fish.style.transform = '';
          showGameOverModal(defeatedTeam === 'red' ? 'blue' : 'red');
        }, 700);
      }, 550);
    }

    function showGameOverModal(winnerSquad) {
      const isRed = winnerSquad === 'red';
      document.getElementById('winnerTitle').innerText = isRed ? '🏆 RED SQUAD WINS!' : '🏆 BLUE SQUAD WINS!';
      document.getElementById('winnerTitle').style.color = isRed ? '#FF385C' : '#2B7FFF';
      document.getElementById('gameOverModal').classList.remove('hidden');
      broadcastPayload({ type: 'SYNC_STATE', status: 'GAMEOVER', buzzedTeam: null });
    }

    function restartGame() {
      SFX.correct();
      document.getElementById('gameOverModal').classList.add('hidden');

      gameState.redBalloons = 5;
      gameState.blueBalloons = 5;
      gameState.qIndex = 0;
      gameState.revealed = false;
      unlockBuzz();
      renderScreen();
      startTimer();

      // 通知全场手机端恢复并重选队伍
      broadcastPayload({ type: 'RESET_GAME' });
    }

    function toggleReveal() {
      SFX.tap();
      gameState.revealed = !gameState.revealed;
      renderScreen();
    }

    function speakCurrentWord() {
      const cur = questions[gameState.qIndex];
      if ('speechSynthesis' in window) {
        const u = new SpeechSynthesisUtterance(cur.hanzi);
        u.lang = 'zh-CN';
        u.rate = 0.85;
        window.speechSynthesis.speak(u);
      }
    }

    function navigateQuestion(step) {
      SFX.tap();
      const nextIdx = gameState.qIndex + step;
      if (nextIdx >= questions.length) {
        const winner = gameState.redBalloons >= gameState.blueBalloons ? 'red' : 'blue';
        showGameOverModal(winner);
        return;
      }
      gameState.qIndex = (nextIdx + questions.length) % questions.length;
      gameState.revealed = false;
      unlockBuzz();
      renderScreen();
      startTimer();
    }

    function toggleAudioMaster() {
      SFX.muted = !SFX.muted;
      document.getElementById('audioIcon').innerText = SFX.muted ? '🔇' : '🔊';
    }

    function showMobileStep2() {
      document.getElementById('mobileStepSquad').classList.add('hidden');

    function mobilePressBuzz() {

      broadcastPayload({ type: 'MOBILE_BUZZ', team: mobileSquad });

    function updateMobileBuzzerState(status, buzzedTeam) {
      const btn = document.getElementById('mobileGiantBuzzBtn');
      const icon = document.getElementById('mobileGiantBtnIcon');
      const text = document.getElementById('mobileGiantBtnText');
      const tip = document.getElementById('mobileStatusTip');
      const isRed = mobileSquad === 'red';

      if (status === 'ACTIVE') {
        btn.disabled = false;
        btn.className = `w-full max-w-xs aspect-square rounded-full border-8 border-black flex flex-col items-center justify-center relative select-none transition-all duration-75 text-white active:scale-95 ${
          isRed ? 'bg-[#FF385C] shadow-[0_12px_0_#8A0020]' : 'bg-[#2B7FFF] shadow-[0_12px_0_#003399]'
        }`;
        icon.innerText = '⚡';
        text.innerText = 'TAP TO BUZZ!';
        tip.innerText = 'ROUND IN PROGRESS! TAP TO BUZZ!';
      } else if (status === 'BUZZED') {
        btn.disabled = true;
        btn.className = 'w-full max-w-xs aspect-square rounded-full border-8 border-black flex flex-col items-center justify-center relative select-none bg-slate-600 text-slate-400 shadow-[0_4px_0_#334155] cursor-not-allowed';
        icon.innerText = '🔒';
        text.innerText = buzzedTeam === mobileSquad ? 'BUZZED!' : 'LOCKED!';
        tip.innerText = buzzedTeam === mobileSquad ? 'YOUR SQUAD BUZZED! ANSWER NOW!' : 'OPPONENT BUZZED! WAITING...';
      } else {
        btn.disabled = true;
        btn.className = 'w-full max-w-xs aspect-square rounded-full border-8 border-black flex flex-col items-center justify-center relative select-none bg-slate-600 text-slate-400 shadow-[0_4px_0_#334155] cursor-not-allowed';
        icon.innerText = '⏳';
        text.innerText = 'WAITING...';
        tip.innerText = 'WAITING FOR SCREEN TO START NEXT ROUND...';
      }
    }

    // ---------- 7. 双向消息分发处理器 ----------
    function handleIncomingMessage(msg) {
      if (!msg) return;

      // 手机打招呼 -> 大屏立即补发当前最新状态
      if (msg.type === 'MOBILE_HELLO' && !isMobileRole) {
        broadcastPayload({ type: 'SYNC_STATE', status: gameState.status, buzzedTeam: gameState.buzzedTeam });
      }

      // 手机按铃到达大屏幕
      if (msg.type === 'MOBILE_BUZZ' && !isMobileRole) {
        triggerBuzz(msg.team);
      }

      // 大屏状态到达手机
      if (msg.type === 'SYNC_STATE' && isMobileRole) {
        updateMobileBuzzerState(msg.status, msg.buzzedTeam);
      }

      // 大屏点击【🔄 PLAY AGAIN】-> 手机端清除持久化状态，回归选队 Step 1
      if (msg.type === 'RESET_GAME' && isMobileRole) {
        localStorage.removeItem('balloon_arena_squad');
        mobileSquad = null;
        showMobileStep1();
      }
    }

    }

      SFX.tap();
      document.getElementById('questionModal').classList.add('hidden');
    }
    function renderModalRoundsList() {
      document.getElementById('modalTotalRounds').innerText = questions.length;
      const listEl = document.getElementById('modalRoundsList');
      listEl.innerHTML = '';
      questions.forEach((q, idx) => {
        listEl.innerHTML += `
          <div class="flex items-center justify-between p-2.5 bg-white border-2 border-black rounded-xl text-sm shadow-sm cursor-pointer hover:bg-amber-50"
               onclick="jumpToQuestion(${idx})">
            <div class="flex items-center gap-2">
              <span class="font-black text-amber-600 font-mono">#${idx + 1}</span>
              <strong class="text-xl font-black text-slate-900">${q.hanzi}</strong>
              <span class="text-amber-800 font-mono font-bold">(${q.pinyin})</span>
              <span class="text-slate-500 font-bold text-xs">- ${q.meaning}</span>
            </div>
            <button onclick="event.stopPropagation(); deleteQuestionItem(${idx})" class="text-rose-500 hover:text-rose-700 font-black p-1 text-base">🗑</button>
          </div>
        `;
      });
    }
    function handleAddWord() {
      const input = document.getElementById('smartInput').value.trim();
      if (!input) return;
      const lower = input.toLowerCase();
      let matched = BUILTIN_DICTIONARY.find(item => item.hanzi === input || item.meaning.toLowerCase().includes(lower));
      let newQ = matched ? { ...matched } : { hanzi: input, pinyin: '—', meaning: 'Custom Entry', collocation: `${input} 常用搭配` };
      questions.push(newQ);
      localStorage.setItem('hsk_supercell_bank', JSON.stringify(questions));
      renderModalRoundsList();
      document.getElementById('smartInput').value = '';
      SFX.correct();
    }
    function deleteQuestionItem(idx) {
      SFX.tap();
      questions.splice(idx, 1);
      localStorage.setItem('hsk_supercell_bank', JSON.stringify(questions));
      if (gameState.qIndex >= questions.length) gameState.qIndex = Math.max(0, questions.length - 1);
      renderModalRoundsList();
    }
    function jumpToQuestion(idx) {
      gameState.qIndex = idx;
      closeQuestionModal();
      gameState.revealed = false;
      unlockBuzz();
      renderScreen();
      startTimer();
    }
    function loadDefaultPresetQuestions() {
      SFX.correct();
      questions = [...BUILTIN_DICTIONARY.slice(0, 5)];
      localStorage.setItem('hsk_supercell_bank', JSON.stringify(questions));
      renderModalRoundsList();
    }
    function clearAllQuestions() {
      if (confirm('确认清空所有题目？')) {
        questions = [];
        localStorage.setItem('hsk_supercell_bank', JSON.stringify(questions));
        renderModalRoundsList();
      }
    }
    function applyAndCloseModal() {
      SFX.correct();
      gameState.redBalloons = 5;
      gameState.blueBalloons = 5;
      gameState.qIndex = 0;
      gameState.revealed = false;
      unlockBuzz();
      closeQuestionModal();
      renderScreen();
      startTimer();
    }

    // 全局快捷键
    window.addEventListener('keydown', (e) => {
      if (isMobileRole) return;
      if (['input', 'textarea'].includes(document.activeElement.tagName.toLowerCase())) return;
      if (e.repeat) return;
      const key = e.key.toUpperCase();
      if (key === 'A') triggerBuzz('red');
      else if (key === 'L') triggerBuzz('blue');
      else if (key === ' ') { e.preventDefault(); toggleReveal(); }
      else if (e.key === 'ArrowRight') navigateQuestion(1);
      else if (e.key === 'ArrowLeft') navigateQuestion(-1);
    });

    window.addEventListener('click', () => SFX.init(), { once: true });

      initNetworkComm();
      if (isMobileRole) {
        initMobileMode();
      } else {
        initScreenMode();
      }
    });
